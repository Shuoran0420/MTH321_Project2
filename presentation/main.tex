\section{Problem}
% ---------------------------------------------------------------------

\begin{frame}{Problem and engineering goal}
  \begin{columns}[T]
    \begin{column}{0.50\textwidth}
      \begin{block}{What we need}
        \begin{itemize}
          \item Compute the quasistatic capacitance per unit length.
          \item Reuse the same mesh for the all-air capacitance.
          \item Convert the matched pair into $\epseff$ and $\Zzero$.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.46\textwidth}
      \begin{block}{What would not be enough}
        \begin{itemize}
          \item A single FEM number near $50\,\Omega$.
          \item Agreement with an empirical formula only.
          \item Unchecked corner and boundary effects.
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
  \vspace{0.7em}
  \begin{alertblock}{Takeaway}
    The result matters because the numerical evidence chain makes it defensible.
  \end{alertblock}
\end{frame}

\begin{frame}{Geometry and PDE model}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \includegraphics[width=\linewidth]{geometry_full_domain.png}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Electrostatic model}
        \[
          -\nabla\cdot(\varepsilon\nabla\phi)=0
        \]
        \[
          \phi=1\,\mathrm{V}\ \text{on trace},\qquad
          \phi=0\ \text{on ground}
        \]
        Homogeneous Neumann conditions are used on the top and side artificial boundaries.
      \end{block}
      \begin{block}{Physical dimensions}
        $H=1.6\,\mathrm{mm}$, $w=3.0\,\mathrm{mm}$,
        $t=0.035\,\mathrm{mm}$, substrate $\epsr=4.4$.
      \end{block}
    \end{column}
  \end{columns}
\begin{alertblock}{Takeaway}A finite 8H domain approximates the open cross-section.\end{alertblock}
\end{frame}

\begin{frame}{Weak form and P1 FEM}
\begin{block}{Weak problem}
Find $\phi\in V_g$ such that $\displaystyle\int_{\Omega}\varepsilon\nabla\phi\cdot\nabla v\,d\Omega=0$ for all $v\in\mathcal V$.
\end{block}
\begin{columns}[T]
\begin{column}{0.48\textwidth}
\begin{block}{Discretisation}
\begin{itemize}\item Conforming triangular $P_1$ elements.\item Trace interior excluded; Dirichlet DOFs eliminated.\item Positive permittivity gives an SPD free system.\end{itemize}
\end{block}
\end{column}
\begin{column}{0.48\textwidth}
\begin{block}{Material interface}
Potential is continuous; flux balance holds in the \textbf{weak sense}. Discrete normal fluxes need not match pointwise on each edge.
\end{block}
\end{column}
\end{columns}
\begin{alertblock}{Takeaway}The weak form handles the material jump; stability and convergence rate are different questions.\end{alertblock}
\end{frame}

\section{Final quantities}

\begin{frame}{Final capacitance selection}
  \begin{columns}[T]
    \begin{column}{0.63\textwidth}
      \includegraphics[width=\linewidth]{final_capacitance_convergence.png}
    \end{column}
    \begin{column}{0.33\textwidth}
      \begin{block}{Selected result}
        Uniform M7, 8H domain
        \[
          C=120.620\,\mathrm{pF/m}
        \]
        M6$\to$M7 change: $0.1556\%$\\
        Preferred QC target: $0.1\%$ (not reached)
      \end{block}
      \begin{block}{Why M7?}
        \begin{itemize}
          \item Monotone capacitance convergence
          \item Stable asymptotic trend
          \item M8 rejected after resource preflight
          \item Richardson extrapolation is diagnostic only
        \end{itemize}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Matched all-air solve and effective permittivity}
  \begin{columns}[T]
    \begin{column}{0.57\textwidth}

    \end{column}
    \begin{column}{0.39\textwidth}
      \begin{block}{Same frozen mesh}
        The all-air solve reuses the exact M7 mesh from the final dielectric case.
      \end{block}
      \begin{block}{Matched capacitance ratio}
        \[
          \Cair=35.9682\,\mathrm{pF/m}
        \]
        \[

      \end{block}
    \end{column}
  \end{columns}
  \begin{alertblock}{Takeaway}
    Mesh identity protects the ratio $C/\Cair$ from a hidden remeshing error.
  \end{alertblock}
\end{frame}

\section{Conclusion}
% ---------------------------------------------------------------------

\begin{frame}{Conclusions and limitations}
\begin{columns}[T]
\begin{column}{0.48\textwidth}
\begin{block}{Evidence and result}
\begin{itemize}
\item Smooth and layered analytical tests check the solver.
\item Triangle overlay corrected the cross-mesh comparison.
\item $C=120.620\,\mathrm{pF/m}$; $\Cair=35.9682\,\mathrm{pF/m}$.
\item $\epseff=3.3535$; $\Zzero=50.642\,\Omega$.
\end{itemize}
\end{block}
\end{column}
\begin{column}{0.48\textwidth}
\begin{block}{Limits}
\begin{itemize}
\item Adaptive refinement saved DOFs, not cumulative time; ideal rate unproven.
\item M7 and finite-domain sensitivities remain.
\item Recorded Apple M2 run: 58 tests, 208 numerical comparisons passed; cross-platform install unverified.
\item Quasi-static model omits losses and radiation.
\end{itemize}
\end{block}
\end{column}
\end{columns}
\begin{alertblock}{Takeaway}A numerical result is credible when its verification and limitations are both visible.\end{alertblock}
\end{frame}
